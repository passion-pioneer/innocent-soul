#!/bin/bash

set -euxo pipefail

# 로그 저장
exec > >(tee -a /var/log/init-script.log)
exec 2>&1

echo "===== INIT SCRIPT START ====="

# Timezone
timedatectl set-timezone Asia/Seoul

# Hostname
hostnamectl set-hostname rocky-web-01

# DNF Cache
dnf makecache

# Update
dnf update -y

# 기본 패키지
dnf install -y \
    vim \
    wget \
    curl \
    unzip \
    git \
    jq \
    net-tools \
    bind-utils \
    tcpdump \
    traceroute \
    lsof \
    tree \
    bash-completion

# 시간 동기화
systemctl enable chronyd
systemctl restart chronyd

echo "===== INIT SCRIPT END ====="
