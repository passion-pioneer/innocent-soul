#!/bin/bash

###############################################################################
# Rocky Linux 8.x Initialization Script
# Purpose : NCP Landing Zone Base Configuration
###############################################################################

set -euxo pipefail

# Redirect all output to log file
exec > >(tee -a /var/log/init-script.log)
exec 2>&1

echo "========== Initialization Started =========="

###############################################################################
# Configure Hostname
###############################################################################

hostnamectl set-hostname cbdcrocky01

###############################################################################
# Configure Time Zone
###############################################################################

timedatectl set-timezone Asia/Seoul

###############################################################################
# Create Administrative Group
###############################################################################

if ! getent group rockyadmin >/dev/null; then
    groupadd -g 2000 rockyadmin
fi

###############################################################################
# Create Administrative User
###############################################################################

if ! id rockyadmin >/dev/null 2>&1; then
    useradd \
        -u 2000 \
        -g rockyadmin \
        -m \
        -d /home/rockyadmin \
        -s /bin/bash \
        rockyadmin
fi

###############################################################################
# Set Initial Password
###############################################################################

echo "rockyadmin:ChangeMe123!" | chpasswd

###############################################################################
# Grant Sudo Privileges
###############################################################################

echo "rockyadmin ALL=(ALL) ALL" > /etc/sudoers.d/rockyadmin
chmod 440 /etc/sudoers.d/rockyadmin

###############################################################################
# Disable Direct Root SSH Login
###############################################################################

sed -i 's/^#\?PermitRootLogin.*/PermitRootLogin no/' /etc/ssh/sshd_config

###############################################################################
# Enable Password Authentication
###############################################################################

sed -i 's/^#\?PasswordAuthentication.*/PasswordAuthentication yes/' /etc/ssh/sshd_config

###############################################################################
# Enable Public Key Authentication
###############################################################################

sed -i 's/^#\?PubkeyAuthentication.*/PubkeyAuthentication yes/' /etc/ssh/sshd_config

###############################################################################
# Restart SSH Service
###############################################################################

systemctl restart sshd

echo "========== Initialization Completed =========="
